cmake_minimum_required(VERSION 3.20)
project(leadcut-thumbnail LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(LeadCutThumbnail SHARED src/provider.cpp exports.def)
target_compile_definitions(LeadCutThumbnail PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(LeadCutThumbnail PRIVATE /EHsc /W4 /utf-8)
target_link_libraries(LeadCutThumbnail PRIVATE windowsapp windowscodecs crypt32 shlwapi ole32 gdi32)
add_executable(thumbnail-probe src/probe.cpp)
target_compile_definitions(thumbnail-probe PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(thumbnail-probe PRIVATE /EHsc /W4 /utf-8)
target_link_libraries(thumbnail-probe PRIVATE ole32 shell32 shlwapi gdi32 windowscodecs)
